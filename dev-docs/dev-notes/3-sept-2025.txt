class chatGlobalVars {
  // Private fields (only accessible inside the class)
  #secretToken;
  #lastSeen;

  constructor(parameters) {
    // Public fields
    this.username = parameters.username || 'Guest';
    this.isLoggedIn = parameters.isLoggedIn || false;
    this.chatRoom = parameters.chatRoom || 'general';
    this.messageCount = 0;

    // Initialize private fields
    this.#secretToken = parameters.secretToken || 'default-token';
    this.#lastSeen = new Date();
  }

  // Public method to send a message
  sendMessage(message) {
    console.log(`${this.username} says: ${message}`);
    this.messageCount++;
    this.#updateLastSeen();
  }

  // Public method to get user status
  getStatus() {
    return `${this.username} is ${this.isLoggedIn ? 'online' : 'offline'} in ${this.chatRoom}`;
  }

  // Private method (only usable inside the class)
  #updateLastSeen() {
    this.#lastSeen = new Date();
  }

  // Public method to get last seen time
  getLastSeen() {
    return this.#lastSeen;
  }

  // Public method to get the secret token (if needed)
  getToken() {
    return this.#secretToken;
  }
}
